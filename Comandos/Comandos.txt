# =====================================================================
# COMANDOS DE DEMOSTRACION EN ORDEN DE EJECUCION
# =====================================================================

# 1. VER CONTEO INICIAL
docker exec -it db-primary psql -U admin_db -d banco_telemetria -c "SELECT count(*) AS total_transacciones FROM transacciones;"

# 2. GENERAR BACKUP FULL
docker exec db-primary mkdir -p /backup_storage/full_backups
docker exec -e PGPASSWORD=postgres123 db-primary pg_dump -U admin_db -d banco_telemetria -F c -b -f /backup_storage/full_backups/full_backup_base.dump
docker exec -it db-primary ls -lh /backup_storage/full_backups/full_backup_base.dump

# 3. INSERTAR 10,000 TRANSACCIONES NUEVAS (POST-BACKUP)
docker exec -it db-primary psql -U admin_db -d banco_telemetria -c "INSERT INTO transacciones (cuenta_id, tipo_transaccion, monto, descripcion, estado) SELECT floor(random() * 90000 + 100001)::INT, (ARRAY['TRANSFERENCIA','DEPOSITO','RETIRO','PAGO'])[floor(random()*4)+1], round((random() * 2500 + 10)::numeric, 2), 'Transaccion critica post-backup #' || s, 'COMPLETADO' FROM generate_series(1, 10000) AS s;"
docker exec -it db-primary psql -U admin_db -d banco_telemetria -c "SELECT count(*) AS total_transacciones FROM transacciones;"

# 4. GENERAR BACKUP INCREMENTAL (DELTA) Y SELLAR WAL
docker exec db-primary mkdir -p /backup_storage/incremental_backups
docker exec db-primary chmod 777 /backup_storage/incremental_backups
docker exec -e PGPASSWORD=postgres123 db-primary psql -U admin_db -d banco_telemetria -c "COPY (SELECT * FROM transacciones WHERE descripcion LIKE 'Transaccion critica post-backup%') TO '/backup_storage/incremental_backups/backup_diferencial.sql';"
docker exec -it db-primary psql -U admin_db -d banco_telemetria -c "SELECT pg_switch_wal();"
docker exec -it db-primary ls -lh /backup_storage/full_backups/full_backup_base.dump /backup_storage/incremental_backups/backup_diferencial.sql

# 5. VER WAL EN VIVO
docker exec -it db-primary ls -lht /backup_storage/wal_archive

# 6. CAUSAR DESASTRE (BORRADO TOTAL DE DATOS)
docker exec -e PGPASSWORD=postgres123 db-primary psql -U admin_db -d banco_telemetria -c "TRUNCATE TABLE transacciones;"
docker exec -it db-primary psql -U admin_db -d banco_telemetria -c "SELECT count(*) FROM transacciones;"

# 7. RESTAURAR SOLO FULL (MUESTRA 400,000 - FALTAN 10,000)
docker exec -e PGPASSWORD=postgres123 db-primary pg_restore -U admin_db -d banco_telemetria --data-only -t transacciones /backup_storage/full_backups/full_backup_base.dump
docker exec -it db-primary psql -U admin_db -d banco_telemetria -c "SELECT count(*) AS total_transacciones FROM transacciones;"

# 8. RESTAURAR INCREMENTAL (MUESTRA 410,000 - 100% RECUPERADO)
docker exec -e PGPASSWORD=postgres123 db-primary psql -U admin_db -d banco_telemetria -c "COPY transacciones FROM '/backup_storage/incremental_backups/backup_diferencial.sql';"
docker exec -it db-primary psql -U admin_db -d banco_telemetria -c "SELECT count(*) AS total_transacciones FROM transacciones;"
